from collections import deque

def remove(n):
    queue.append(n)
    while queue:
        idx = queue.popleft()
        coins[idx] = -1
        if idx > 0 and coins[idx-1] != -1:
            coins[idx-1] = 1-coins[idx-1]
            if coins[idx-1]:
                queue.append(idx-1)
        if idx < N-1 and coins[idx+1] != -1:
            coins[idx+1] = 1-coins[idx+1]
            if coins[idx+1]:
                queue.append(idx+1)
    
    

TC = int(input())

    coins = list(map(int, list(input())))
    N = len(coins)
    queue = deque()

        if coins[i] == 1:
