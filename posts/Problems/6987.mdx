### 입력 

 첫째 줄부터 넷째 줄까지 각 줄마다 6개국의 결과가 나라별로 승, 무승부, 패의 순서로 빈칸을 하나 사이에 두고 18개의 숫자로 주어진다. 승, 무, 패의 수는 6보다 작거나 같은 자연수 또는 0이다.

### 출력 

 입력에서 주어진 네 가지 결과에 대하여 가능한 결과는 1, 불가능한 결과는 0을 빈칸을 하나 사이에 두고 출력한다.

### 풀이
```java
import java.util.*;
import java.io.*;

public class Main {
  static int[][] games;
  static boolean answer;
  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st;
    StringBuilder sb = new StringBuilder();

    for (int t=0; t<4; t++){
      games = new int[6][3];
      answer = false;
      st = new StringTokenizer(br.readLine());
      int sum = 0;
      for(int i=0; i<6; i++){
        for (int j=0; j<3; j++){
          games[i][j] = Integer.parseInt(st.nextToken());
          sum += games[i][j];
        }
      }
      if(sum == 30) backTrack(0);
      sb.append(answer? 1 : 0).append(" ");
    }
    System.out.println(sb);
  }

  private static void backTrack(int match){
    if(answer) return;
    if(match == 15){
      answer = true;
      return;
    }

    int[] team = matchToTeam(match);
    int x = team[0], y = team[1];
    // x승
    if(games[x][0]>0 && games[y][2]>0){
      games[x][0]--;
      games[y][2]--;
      backTrack(match+1);
      games[x][0]++;
      games[y][2]++;
    }
    // 무
    if(games[x][1]>0 && games[y][1]>0){
      games[x][1]--;
      games[y][1]--;
      backTrack(match+1);
      games[x][1]++;
      games[y][1]++;
    }

    // x패
    if(games[x][2]>0 && games[y][0]>0){
      games[x][2]--;
      games[y][0]--;
      backTrack(match+1);
      games[x][2]++;
      games[y][0]++;
    }
  }

  private static int[] matchToTeam(int match){
    int cnt = 0;
    for(int i=0; i<5; i++){
      for (int j=i+1; j<6; j++){
        if(cnt == match) return new int[] {i, j};
        cnt++;
      }
    }
    return new int[] {-1, -1};
  }
}
```
